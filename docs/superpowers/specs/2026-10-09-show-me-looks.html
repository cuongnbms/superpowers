def do_GET(self):
    path, _, query = self.path.partition("?")
    if not check_key(self, query):
        return self.send_error(403)
    if path == "/api/screens":
        return send_json(self, list_screens(SCREENS))
def check_key(req, query):
    given = parse_qs(query).get("key", [None])[0]
    given = given or cookie_value(req, "show_me_key")
    return hmac.compare_digest(given or "", KEY)
def list_screens(root):
    files = [p for p in root.glob("*.html") if not p.name.startswith(".")]
    files.sort(key=lambda p: p.stat().st_mtime)
    return [{"name": p.name, "mtime": p.stat().st_mtime} for p in files]
def send_json(req, screens):
    newest = screens[-1]["name"] if screens else None
    body = json.dumps({"screens": screens, "newest": newest})
    req.send_response(200)
    req.send_header("Content-Type", "application/json")
    req.end_headers()
    req.wfile.write(body.encode())
